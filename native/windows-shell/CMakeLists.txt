cmake_minimum_required(VERSION 3.20)
project(smolpng_shell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_executable(shell-contract-test core_test.cpp)
add_test(NAME shell-contract COMMAND shell-contract-test)
if(WIN32)
  add_library(SmolpngShell SHARED shell.cpp SmolpngShell.def)
  target_compile_definitions(SmolpngShell PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(SmolpngShell PRIVATE shell32 shlwapi ole32 uuid)
  if(MSVC)
    target_compile_options(SmolpngShell PRIVATE /W4 /EHsc /guard:cf /utf-8)
    target_link_options(SmolpngShell PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
  endif()
endif()
